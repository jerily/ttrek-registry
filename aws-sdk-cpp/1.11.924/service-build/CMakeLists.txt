cmake_minimum_required(VERSION 3.22)
project(aws_service LANGUAGES C CXX)

if(NOT AWS_SDK_SOURCE_DIR OR NOT AWS_SDK_SERVICE)
    message(FATAL_ERROR "Set AWS_SDK_SOURCE_DIR and AWS_SDK_SERVICE")
endif()
file(STRINGS "${AWS_SDK_SOURCE_DIR}/src/aws-cpp-sdk-core/include/aws/core/VersionConfig.h"
    version_line LIMIT_COUNT 1 REGEX "AWS_SDK_VERSION_STRING.*[0-9]+\\.[0-9]+\\.[0-9]+")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" VERSION_STRING "${version_line}")
find_package(ZLIB REQUIRED)
find_package(AWSSDK ${VERSION_STRING} EXACT REQUIRED)
get_dependencies_for_sdk(${AWS_SDK_SERVICE} service_dependencies)
string(REPLACE "," ";" service_dependencies "${service_dependencies}")
list(FILTER service_dependencies EXCLUDE REGEX "^$")
foreach(dependency IN LISTS service_dependencies)
    find_package(aws-cpp-sdk-${dependency} ${VERSION_STRING} EXACT REQUIRED CONFIG)
endforeach()
get_target_property(core_type aws-cpp-sdk-core TYPE)
if((BUILD_SHARED_LIBS AND NOT core_type STREQUAL "SHARED_LIBRARY") OR
   (NOT BUILD_SHARED_LIBS AND NOT core_type STREQUAL "STATIC_LIBRARY"))
    message(FATAL_ERROR "Service and installed core must use the same BUILD_SHARED_LIBS setting")
endif()

set(AWS_NATIVE_SDK_ROOT "${AWS_SDK_SOURCE_DIR}")
list(APPEND CMAKE_MODULE_PATH "${AWS_NATIVE_SDK_ROOT}/cmake")
set(SIMPLE_INSTALL ON)
set(ENABLE_UNITY_BUILD ON)
set(ENABLE_RTTI ON)
set(ENABLE_VIRTUAL_OPERATIONS ON)
set(AWS_SDK_WARNINGS_ARE_ERRORS ON)
set(CPP_STANDARD 11)
include(initialize_project_version)
include(resolve_platform)
if(APPLE)
    set(CMAKE_INSTALL_RPATH "@loader_path")
elseif(UNIX)
    set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
include(CMakePackageConfigHelpers)
include(utilities)
include(compiler_settings)

if(AWS_SDK_SERVICE STREQUAL "transfer")
    set(service_directory "src/aws-cpp-sdk-transfer")
else()
    set(service_directory "generated/src/aws-cpp-sdk-${AWS_SDK_SERVICE}")
endif()
add_subdirectory("${AWS_SDK_SOURCE_DIR}/${service_directory}" service)
